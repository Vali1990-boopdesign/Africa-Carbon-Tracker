// ============================================
// ENGAGEMENT TRACKING
// ============================================

// Track time-based engagement
let engagementTime = 0;
const engagementThresholds = {
  30: false,
  120: false,  // 2 minutes
  300: false   // 5 minutes
};

setInterval(() => {
  engagementTime += 10;
  
  Object.keys(engagementThresholds).forEach(threshold => {
    if (engagementTime >= threshold && !engagementThresholds[threshold]) {
      gtag('event', 'time_engaged', {
        'engagement_duration': threshold,
        'engagement_level': threshold >= 120 ? 'high' : 'medium'
      });
      engagementThresholds[threshold] = true;
    }
  });
}, 10000); // Check every 10 seconds

// Track if user actually explored data (any interaction)
let hasInteracted = false;
function markUserAsEngaged() {
  if (!hasInteracted) {
    gtag('event', 'data_explored', {
      'first_interaction_time': engagementTime
    });
    hasInteracted = true;
  }
}

// ============================================
// FILTER TRACKING
// ============================================

function trackFilterUse(filterType, filterValue, resultCount) {
  gtag('event', 'filter_applied', {
    'filter_type': filterType,        // e.g., "country", "sector", "credit_type"
    'filter_value': filterValue,       // e.g., "Kenya", "Renewable Energy"
    'result_count': resultCount,       // How many results after filtering
    'event_category': 'interaction'
  });
  markUserAsEngaged();
}

document.querySelector('#country-filter').addEventListener('change', function(e) {

// CHART/VISUALIZATION TRACKING
// ============================================

function trackChartInteraction(chartType, interactionType, dataPoint) {
  gtag('event', 'chart_interaction', {
    'chart_type': chartType,           // e.g., "map", "bar_chart", "timeline"
    'interaction_type': interactionType, // e.g., "click", "hover", "zoom"
    'data_point': dataPoint,           // Optional: what they clicked on
    'event_category': 'visualization'
  });
  markUserAsEngaged();
}

function trackSearch(searchTerm, resultCount) {
  gtag('event', 'search', {
    'search_term': searchTerm,
    'result_count': resultCount,
    'event_category': 'interaction'
  });
  markUserAsEngaged();
}

  e.preventDefault();

  trackSearch(query, results.length);
});

// ============================================
// DATA POINT SELECTION (KEY EVENT)
// ============================================

function trackDataPointSelection(itemType, itemName, itemDetails) {
  gtag('event', 'data_point_selected', {
    'item_type': itemType,             // e.g., "project", "credit", "organization"
    'item_name': itemName,             // e.g., "Kenya Wind Farm Project"
    'item_id': itemDetails.id,
    'country': itemDetails.country,
    'sector': itemDetails.sector,
    'event_category': 'engagement'
  });
  markUserAsEngaged();
}

// Example: When user clicks on a specific carbon project
document.querySelectorAll('.project-card').forEach(card => {
  card.addEventListener('click', function() {
    const projectData = JSON.parse(this.dataset.project);
    
    trackDataPointSelection('project', projectData.name, {
      id: projectData.id,
      country: projectData.country,
      sector: projectData.sector
    });
  });
});

function trackComparison(items) {
  gtag('event', 'comparison_made', {
    'item_count': items.length,
    'comparison_type': 'projects',     // or "credits", "countries"
    'items_compared': items.join(', '),
    'event_category': 'advanced_use'
  });
}
