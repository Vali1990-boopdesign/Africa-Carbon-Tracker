    gtag: (...args: any[]) => void;

// ============================================
// ENGAGEMENT TRACKING
// ============================================

let engagementTime = 0;
const engagementThresholds: { [key: number]: boolean } = {
  30: false,
  120: false,
  300: false
};

export function initEngagementTracking() {
  setInterval(() => {
    engagementTime += 10;
    
    Object.keys(engagementThresholds).forEach((threshold) => {
      const thresholdNum = parseInt(threshold);
      if (engagementTime >= thresholdNum && !engagementThresholds[thresholdNum]) {
        trackEvent('time_engaged_custom', {
          engagement_duration: thresholdNum,
          engagement_level: thresholdNum >= 120 ? 'high' : 'medium'
        });
        engagementThresholds[thresholdNum] = true;
      }
    });
  }, 10000);
}

// ============================================
// BASE TRACKING FUNCTION
// ============================================

export function trackEvent(eventName: string, params?: Record<string, any>) {
  if (typeof window !== 'undefined' && window.gtag) {
    window.gtag('event', eventName, params);
    console.log('📊 GA4 Event:', eventName, params); // Debug logging
  } else {
    console.warn('GA4 not loaded yet');
  }
}

// ============================================
// SPECIFIC EVENT TRACKERS
// ============================================

export function trackFilterApplied(
  filterType: string, 
  filterValue: string, 
  resultCount: number
) {
  trackEvent('filter_applied', {
    filter_type: filterType,
    filter_value: filterValue,
    result_count: resultCount,
    event_category: 'interaction'
  });
}

export function trackChartInteraction(
  chartType: string, 
  interactionType: string, 
  dataPoint?: string
) {
  trackEvent('chart_interaction', {
    chart_type: chartType,
    interaction_type: interactionType,
    data_point: dataPoint,
    event_category: 'visualization'
  });
}

export function trackDataPointSelected(
  itemType: string,
  itemName: string,
  itemDetails: {
    id?: string;
    country?: string;
    sector?: string;
  }
) {
  trackEvent('data_point_selected', {
    item_type: itemType,
    item_name: itemName,
    item_id: itemDetails.id,
    country: itemDetails.country,
    sector: itemDetails.sector,
    event_category: 'engagement'
  });
}

export function trackSearch(searchTerm: string, resultCount: number) {
  trackEvent('search_performed', {
    search_term: searchTerm,
    result_count: resultCount,
    event_category: 'interaction'
  });
}

export function trackExternalLink(linkUrl: string, linkContext: string) {
  trackEvent('external_link_clicked', {
    link_url: linkUrl,
    link_context: linkContext,
    event_category: 'conversion'
  });
}

export function trackFullReportView(projectName: string, projectId: string) {
  trackEvent('full_report_viewed', {
    project_name: projectName,
    project_id: projectId,
    event_category: 'conversion',
    value: 1
  });
}

export function trackContactClick(contactType: string, projectId?: string) {
  trackEvent('contact_clicked', {
    contact_type: contactType,
    project_id: projectId,
    event_category: 'conversion',
    value: 5
  });
}

// ============================================
// REFERRER TRACKING
// ============================================

export function trackReferrer() {
  const referrer = document.referrer;
  
  if (referrer.includes('thecatalystfund.com')) {
    trackEvent('arrived_from_catalyst_fund', {
      referrer_url: referrer,
      entry_source: 'carbon_modules_report'
    });
  }
  
  trackEvent('dashboard_loaded', {
    entry_source: referrer || 'direct',
    timestamp: new Date().toISOString()
  });
}
